<?php
defined('BASEPATH') OR exit('No direct script access allowed');

class Supplier extends CI_Controller {

    public function __construct()
    {
        parent::__construct();
        $this->load->database();
        header("Content-Type: application/json");
    }

    /**
     * ✅ LIST (Already yours, optimized)
     */
    public function list()
    {
        $org = $this->input->get('orgID');

        if(!$org){
            echo json_encode([
                "status" => false,
                "message" => "orgID is required"
            ]);
            return;
        }

        $data = $this->db->query("
            SELECT
                b.*,
                d.district_name,
                p.police_station_name,
                IFNULL(sf.opening_dr, 0) AS opening_dr,
                IFNULL(sf.opening_cr, 0) AS opening_cr
            FROM z_str_supplier b
            LEFT JOIN z_sys_district d ON d.id = b.district
            LEFT JOIN z_sys_police_station p ON p.id = b.police_station_thana
            LEFT JOIN z_str_supplier_finance sf ON sf.supplier_id = b.id
            WHERE b.orgID = ?
        ", [$org])->result();

        echo json_encode([
            "status" => true,
            "data" => $data
        ]);
    }

    /**
     * ✅ DETAILS
     * GET: /api/Supplier/details?id=1
     */
    public function details()
    {
        $id = $this->input->get('id');

        if(!$id){
            echo json_encode([
                "status" => false,
                "message" => "ID required"
            ]);
            return;
        }

        $data = $this->db->get_where('z_str_supplier', ['id' => $id])->row();

        echo json_encode([
            "status" => true,
            "data" => $data
        ]);
    }

    /**
     * ✅ CREATE (Same as your web logic)
     */
    public function create()
    {
        $input = json_decode(file_get_contents("php://input"), true);

        if(empty($input['name']) || empty($input['orgID'])){
            echo json_encode([
                "status" => false,
                "message" => "name & orgID required"
            ]);
            return;
        }

        $this->db->trans_start(); // ✅ transaction start

        // ✅ 1. Supplier Insert
        $supplierData = [
            'name' => $input['name'],
            'contact_person' => $input['contact_person'] ?? '',
            'contact' => $input['contact'] ?? '',
            'email' => $input['email'] ?? '',
            'facebook_page_link' => $input['facebook_page_link'] ?? '',
            'district' => $input['district'] ?? null,
            'police_station_thana' => $input['police_station_thana'] ?? null,
            'supplier_category' => $input['supplier_category'] ?? null,
            'address' => $input['address'] ?? '',
            'comments' => $input['comments'] ?? '',
            'status' => 1,
            'create_date' => date('Y-m-d'),
            'create_by' => $input['user_id'] ?? 1,
            'orgID' => $input['orgID']
        ];

        $this->db->insert('z_str_supplier', $supplierData);
        $supplier_id = $this->db->insert_id();

        // ✅ 2. Finance Insert (IMPORTANT FIX)
        if(isset($input['opening_dr']) || isset($input['opening_cr'])){
            $financeData = [
                'supplier_id'   => $supplier_id,
                'opening_dr'  => $input['opening_dr'] ?? 0,
                'opening_cr'  => $input['opening_cr'] ?? 0,
                'create_date' => date('Y-m-d H:i:s'),
                'create_by'   => $input['user_id'] ?? 1,
            ];
            $this->db->insert('z_str_supplier_finance', $financeData);
        }

        $this->db->trans_complete(); // ✅ transaction end

        if ($this->db->trans_status() === FALSE) {
            echo json_encode([
                "status" => false,
                "message" => "Insert failed"
            ]);
        } else {
            echo json_encode([
                "status" => true,
                "message" => "Supplier created successfully",
                "supplier_id" => $supplier_id
            ]);
        }
    }

    /**
     * ✅ UPDATE
     * POST: /api/Supplier/update?id=1
     */
    public function update()
    {
        $id = $this->input->get('id');
        $input = json_decode(file_get_contents("php://input"), true);

        if(!$id){
            echo json_encode([
                "status" => false,
                "message" => "ID required"
            ]);
            return;
        }

        $this->db->trans_start(); // ✅ transaction start

        // ✅ 1. Supplier Update
        $supplierData = [
            'name' => $input['name'],
            'contact_person' => $input['contact_person'] ?? '',
            'contact' => $input['contact'] ?? '',
            'email' => $input['email'] ?? '',
            'facebook_page_link' => $input['facebook_page_link'] ?? '',
            'district' => $input['district'] ?? null,
            'police_station_thana' => $input['police_station_thana'] ?? null,
            'supplier_category' => $input['supplier_category'] ?? null,
            'address' => $input['address'] ?? '',
            'comments' => $input['comments'] ?? '',
            'status' => $input['status'] ?? 1,
            'update_date' => date('Y-m-d'),
            'update_by' => $input['user_id'] ?? 1,
        ];

        $this->db->where('id', $id);
        $this->db->update('z_str_supplier', $supplierData);

        // ✅ 2. Finance Update (IMPORTANT FIX)
        if(isset($input['opening_dr']) || isset($input['opening_cr'])){
            // Check if finance record exists
            $this->db->where('supplier_id', $id);
            $existingFinance = $this->db->get('z_str_supplier_finance')->row();

            $financeData = [
                'opening_dr' => $input['opening_dr'] ?? 0,
                'opening_cr' => $input['opening_cr'] ?? 0,
                'update_date' => date('Y-m-d H:i:s'),
                'update_by' => $input['user_id'] ?? 1,
            ];

            if($existingFinance){
                // Update existing finance record
                $this->db->where('supplier_id', $id);
                $this->db->update('z_str_supplier_finance', $financeData);
            } else {
                // Insert new finance record
                $financeData['supplier_id'] = $id;
                $financeData['create_date'] = date('Y-m-d H:i:s');
                $financeData['create_by'] = $input['user_id'] ?? 1;
                $this->db->insert('z_str_supplier_finance', $financeData);
            }
        }

        $this->db->trans_complete(); // ✅ transaction end

        if ($this->db->trans_status() === FALSE) {
            echo json_encode([
                "status" => false,
                "message" => "Update failed"
            ]);
        } else {
            echo json_encode([
                "status" => true,
                "message" => "Supplier updated successfully"
            ]);
        }
    }

    /**
     * ✅ DELETE
     * GET: /api/Supplier/delete?id=1
     */
    public function delete()
    {
        $id = $this->input->get('id');

        if(!$id){
            echo json_encode([
                "status" => false,
                "message" => "ID required"
            ]);
            return;
        }

        $this->db->where('id', $id);
        $this->db->delete('z_str_supplier');

        echo json_encode([
            "status" => true,
            "message" => "Supplier deleted successfully"
        ]);
    }
}
