<?php
defined('BASEPATH') OR exit('No direct script access allowed');

class CashBook extends CI_Controller {

    public function __construct() {
        parent::__construct();
        header("Content-Type: application/json");
    }

    public function list() {

        header("Content-Type: application/json");

        /*
        |--------------------------------------------------------------------------
        | Get Request Data
        |--------------------------------------------------------------------------
        */

        $orgID = $this->input->get_post('orgID');
        $start = $this->input->get_post('start');
        $end   = $this->input->get_post('end');

        /*
        |--------------------------------------------------------------------------
        | Validation
        |--------------------------------------------------------------------------
        */

        if (!$orgID) {
            echo json_encode([
                'status' => false,
                'message' => 'orgID is required'
            ]);
            return;
        }

        /*
        |--------------------------------------------------------------------------
        | Default Date
        |--------------------------------------------------------------------------
        */

        if (!$start) {
            $start = date('Y-m-d');
        }

        if (!$end) {
            $end = date('Y-m-d');
        }

        // Full datetime format
        $startDate = $start . ' 00:00:00';
        $endDate   = $end . ' 23:59:59';

        /*
        |--------------------------------------------------------------------------
        | Opening Cash Balance
        |--------------------------------------------------------------------------
        */

        $accOpenCash = $this->db->query("
            SELECT 
                SUM(c.amount) as amount
            FROM z_acc_opening_balance c
            WHERE c.orgID = ?
        ", [$orgID])->row();

        /*
        |--------------------------------------------------------------------------
        | Previous Income Before Start Date
        |--------------------------------------------------------------------------
        */

        $accOpenIncome = $this->db->query("
            SELECT 
                SUM(i.incomeAmount) as amount
            FROM z_acc_income i
            WHERE i.date < ?
            AND i.orgID = ?
        ", [$startDate, $orgID])->row();

        /*
        |--------------------------------------------------------------------------
        | Previous Expense Before Start Date
        |--------------------------------------------------------------------------
        */

        $accOpenExpense = $this->db->query("
            SELECT 
                SUM(e.expenseAmount) as amount
            FROM z_acc_expense e
            WHERE e.date < ?
            AND e.orgID = ?
        ", [$startDate, $orgID])->row();

        /*
        |--------------------------------------------------------------------------
        | Opening Balance Calculation
        |--------------------------------------------------------------------------
        */

        $openingCash    = $accOpenCash->amount ?? 0;
        $openingIncome  = $accOpenIncome->amount ?? 0;
        $openingExpense = $accOpenExpense->amount ?? 0;

        $openingBalance =
            ($openingCash + $openingIncome)
            - $openingExpense;

        /*
        |--------------------------------------------------------------------------
        | Income Summary
        |--------------------------------------------------------------------------
        */

        $incomeSummary = $this->db->query("
            SELECT 
                i.incomeType,
                (
                    SELECT t.categoryName 
                    FROM z_acc_chart t 
                    WHERE i.incomeType = t.id
                ) as head,
                SUM(i.incomeAmount) as amount
            FROM z_acc_income i
            WHERE i.date >= ?
            AND i.date <= ?
            AND i.orgID = ?
            GROUP BY i.incomeType
            ORDER BY i.incomeType ASC
        ", [$startDate, $endDate, $orgID])->result();

        /*
        |--------------------------------------------------------------------------
        | Expense Summary
        |--------------------------------------------------------------------------
        */

        $expenseSummary = $this->db->query("
            SELECT 
                e.expenseType,
                (
                    SELECT t.categoryName 
                    FROM z_acc_chart t 
                    WHERE e.expenseType = t.id
                ) as head,
                SUM(e.expenseAmount) as amount
            FROM z_acc_expense e
            WHERE e.date >= ?
            AND e.date <= ?
            AND e.orgID = ?
            GROUP BY e.expenseType
            ORDER BY e.expenseType ASC
        ", [$startDate, $endDate, $orgID])->result();

        /*
        |--------------------------------------------------------------------------
        | Total Income
        |--------------------------------------------------------------------------
        */

        $totalIncomeRow = $this->db->query("
            SELECT 
                SUM(i.incomeAmount) as total_income
            FROM z_acc_income i
            WHERE i.date >= ?
            AND i.date <= ?
            AND i.orgID = ?
        ", [$startDate, $endDate, $orgID])->row();

        /*
        |--------------------------------------------------------------------------
        | Total Expense
        |--------------------------------------------------------------------------
        */

        $totalExpenseRow = $this->db->query("
            SELECT 
                SUM(e.expenseAmount) as total_expense
            FROM z_acc_expense e
            WHERE e.date >= ?
            AND e.date <= ?
            AND e.orgID = ?
        ", [$startDate, $endDate, $orgID])->row();

        $totalIncome  = $totalIncomeRow->total_income ?? 0;
        $totalExpense = $totalExpenseRow->total_expense ?? 0;

        /*
        |--------------------------------------------------------------------------
        | Closing Balance
        |--------------------------------------------------------------------------
        */

        $closingBalance =
            $openingBalance
            + $totalIncome
            - $totalExpense;

        /*
        |--------------------------------------------------------------------------
        | Cashbook Transactions
        |--------------------------------------------------------------------------
        */

        $transactions = $this->db->query("
            SELECT 
                'Income' as type,
                (
                    SELECT h.categoryName 
                    FROM z_acc_chart h 
                    WHERE h.id = i.incomeType
                ) as head,
                i.incomeAmount as amount,
                i.comments,
                i.date,
                i.create_date
            FROM z_acc_income i
            WHERE i.orgID = ?
            AND i.date BETWEEN ? AND ?

            UNION ALL

            SELECT 
                'Expense' as type,
                (
                    SELECT h.categoryName 
                    FROM z_acc_chart h 
                    WHERE h.id = e.expenseType
                ) as head,
                e.expenseAmount as amount,
                e.comments,
                e.date,
                e.create_date
            FROM z_acc_expense e
            WHERE e.orgID = ?
            AND e.date BETWEEN ? AND ?

            ORDER BY date ASC
        ", [
            $orgID,
            $startDate,
            $endDate,
            $orgID,
            $startDate,
            $endDate
        ])->result();

        /*
        |--------------------------------------------------------------------------
        | Final Response
        |--------------------------------------------------------------------------
        */

        echo json_encode([
            'status' => true,
            'message' => 'Cash book fetched successfully',

            'start' => $startDate,
            'end' => $endDate,

            'opening_cash' => $openingCash,
            'opening_income' => $openingIncome,
            'opening_expense' => $openingExpense,

            'opening_balance' => $openingBalance,

            'total_income' => $totalIncome,
            'total_expense' => $totalExpense,

            'closing_balance' => $closingBalance,

            'income_summary' => $incomeSummary,
            'expense_summary' => $expenseSummary,

            'transaction_count' => count($transactions),
            'transactions' => $transactions
        ]);
    }
}
